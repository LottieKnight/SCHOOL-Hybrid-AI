# next-intl Platform i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add full UI translation to the SCHOOL web app for all 11 official South African text languages, driven by the user's saved language preference (no URL changes), starting with the 10 highest-traffic pages.

**Architecture:** next-intl with cookie-driven locale resolution. A same-origin Next.js Route Handler (`/api/locale`) sets the `NEXT_LOCALE` cookie when the user saves their language (the Fastify API is cross-origin and cannot set the cookie). `i18n/request.ts` reads the cookie and loads `messages/<locale>.json`. Translations are AI-generated by a Claude API script and stored as version-controlled JSON.

**Tech Stack:** Next.js 16 App Router, next-intl, TypeScript, Vitest, Anthropic SDK (translation script), Fastify (API enum widening).

```ts
import { getRequestConfig } from "next-intl/server";
import { cookies } from "next/headers";
import { DEFAULT_LOCALE, isLocale } from "@/lib/locale";

export default getRequestConfig(async () => {
  const cookieStore = await cookies();
  const cookieValue = cookieStore.get("NEXT_LOCALE")?.value;
  const locale = isLocale(cookieValue) ? cookieValue : DEFAULT_LOCALE;

  // English is always loaded as the fallback source. For non-English locales we
  // merge the locale file over English so any untranslated keys render in English
  // instead of throwing a MISSING_MESSAGE error.
  const english = (await import("../messages/en.json")).default;
  const messages =
    locale === DEFAULT_LOCALE
      ? english
      : { ...english, ...(await import(`../messages/${locale}.json`)).default };

  return { locale, messages };
});
```

```ts
import { NextResponse } from "next/server";
import { isLocale } from "@/lib/locale";

export const runtime = "nodejs";

export async function POST(request: Request) {
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ error: "Invalid JSON body." }, { status: 400 });
  }

  const locale = (body as { locale?: unknown })?.locale;
  if (!isLocale(locale)) {
    return NextResponse.json({ error: "Unsupported locale." }, { status: 400 });
  }

  const response = NextResponse.json({ ok: true, locale });
  response.cookies.set("NEXT_LOCALE", locale, {
    path: "/",
    sameSite: "lax",
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    maxAge: 60 * 60 * 24 * 365, // 1 year
  });
  return response;
}
```

```ts
/**
 * Translates messages/en.json into all target SA languages using the Claude API.
 *
 * - Idempotent: keys already present in a target file are preserved (human
 *   corrections are never overwritten). Only missing keys are translated.
 * - Run: ANTHROPIC_API_KEY=... npx tsx scripts/translate.ts
 */
import Anthropic from "@anthropic-ai/sdk";
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { LOCALES, LOCALE_NAMES, type Locale } from "../lib/locale";

const MESSAGES_DIR = join(import.meta.dirname, "..", "messages");
const MODEL = "claude-sonnet-4-6";

type Messages = Record<string, Record<string, string>>;

function readJson(path: string): Messages {
  try {
    return JSON.parse(readFileSync(path, "utf8"));
  } catch {
    return {};
  }
}

// Returns the subset of `source` whose leaf keys are absent from `target`.
function missingKeys(source: Messages, target: Messages): Messages {
  const out: Messages = {};
  for (const [ns, entries] of Object.entries(source)) {
    for (const [key, value] of Object.entries(entries)) {
      if (target[ns]?.[key] === undefined) {
        (out[ns] ??= {})[key] = value;
      }
    }
  }
  return out;
}

// Deep-merges `incoming` into `base` (incoming wins only for keys it defines).
function mergeMessages(base: Messages, incoming: Messages): Messages {
  const out: Messages = structuredClone(base);
  for (const [ns, entries] of Object.entries(incoming)) {
    out[ns] = { ...(out[ns] ?? {}), ...entries };
  }
  return out;
}

async function translateChunk(
  client: Anthropic,
  languageName: string,
  chunk: Messages,
): Promise<Messages> {
  const system = [
    `You are a professional translator for an educational platform serving South African learners (Grade R-12).`,
    `Translate the JSON string VALUES into ${languageName}. Rules:`,
    `- Return ONLY valid JSON with the exact same key structure. Do not add or remove keys.`,
    `- Preserve any {placeholder} markers EXACTLY (e.g. {name}, {count}) — do not translate or reorder their contents.`,
    `- Use clear, age-appropriate educational register.`,
    `- Keep widely-used English technical terms in English where no established ${languageName} equivalent exists.`,
  ].join("\n");

  const message = await client.messages.create({
    model: MODEL,
    max_tokens: 8000,
    system,
    messages: [{ role: "user", content: JSON.stringify(chunk, null, 2) }],
  });

  const text = message.content
    .filter((block): block is Anthropic.TextBlock => block.type === "text")
    .map((block) => block.text)
    .join("");

  const jsonStart = text.indexOf("{");
  const jsonEnd = text.lastIndexOf("}");
  if (jsonStart === -1 || jsonEnd === -1) {
    throw new Error(`No JSON object found in model response for ${languageName}`);
  }
  return JSON.parse(text.slice(jsonStart, jsonEnd + 1));
}

async function main() {
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!apiKey) {
    console.error("ANTHROPIC_API_KEY is not set.");
    process.exit(1);
  }
  const client = new Anthropic({ apiKey });
  const source = readJson(join(MESSAGES_DIR, "en.json"));

  for (const locale of LOCALES.filter((l): l is Locale => l !== "en")) {
    const targetPath = join(MESSAGES_DIR, `${locale}.json`);
    const existing = readJson(targetPath);
    const todo = missingKeys(source, existing);
    const todoCount = Object.values(todo).reduce((n, e) => n + Object.keys(e).length, 0);

    if (todoCount === 0) {
      console.log(`${locale} (${LOCALE_NAMES[locale]}): up to date — 0 keys.`);
      continue;
    }

    console.log(`${locale} (${LOCALE_NAMES[locale]}): translating ${todoCount} keys...`);
    try {
      const translated = await translateChunk(client, LOCALE_NAMES[locale], todo);
      const merged = mergeMessages(existing, translated);
      writeFileSync(targetPath, JSON.stringify(merged, null, 2) + "\n");
      console.log(`${locale}: wrote ${todoCount} keys.`);
    } catch (error) {
      console.error(`${locale}: FAILED —`, error instanceof Error ? error.message : error);
    }
  }
}

void main();
```

- [ ] **Step 1: Write the test**

Create `apps/web/test/i18n-parity.test.ts`:
```ts
import { describe, expect, it } from "vitest";
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { LOCALES } from "@/lib/locale";

const DIR = join(__dirname, "..", "messages");

function leafKeys(obj: Record<string, Record<string, string>>): Set<string> {
  const keys = new Set<string>();
  for (const [ns, entries] of Object.entries(obj)) {
    for (const key of Object.keys(entries)) keys.add(`${ns}.${key}`);
  }
  return keys;
}

const english = leafKeys(JSON.parse(readFileSync(join(DIR, "en.json"), "utf8")));

describe("i18n message parity", () => {
  for (const locale of LOCALES) {
    it(`${locale}.json contains no keys absent from en.json`, () => {
      const target = leafKeys(JSON.parse(readFileSync(join(DIR, `${locale}.json`), "utf8")));
      const stray = [...target].filter((k) => !english.has(k));
      expect(stray).toEqual([]);
    });
  }
});
```

## Notes for the Implementer

- **Cross-origin reminder:** Never try to set `NEXT_LOCALE` from the Fastify API — it is a different origin and the cookie will not stick on the Next.js domain. The cookie is always set by the same-origin `/api/locale` Route Handler (Task 5).
- **English fallback is intentional:** `i18n/request.ts` merges English under every locale, so a missing translated key renders English rather than throwing. This means Phase 2 pages (not yet translated) keep working in English with zero errors.
- **Endonyms stay untranslated:** Language names in the dropdown ("isiZulu", "Afrikaans") are written in their own language by convention — do not route them through `t()`.
- **Re-running the script is safe:** It only fills missing keys and never overwrites existing values, so human corrections committed to a locale file survive future runs.
